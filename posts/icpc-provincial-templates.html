Graphs and Search

BFS 最短步数

适用：每一步代价相同，求最少操作次数/网格最短路/状态图最短路。

易错点：visited 或 dist 要在入队时标记；第一次到达就是最短步数；状态范围要开够。


#include <bits/stdc++.h>
using namespace std;
const int N = 10010;
int dist[N];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  memset(dist,-1,sizeof dist);
  queue<int> q;
  dist[n] = 0;
  q.push(n);
  while (!q.empty()) {
    int cur = q.front();
    q.pop();
    if (cur == m) {
      cout << dist[cur] << endl;
      return 0;
    }
    if (cur * 2 < N && dist[cur * 2] == -1) {
      dist[cur * 2] = dist[cur] + 1;
      q.push(cur * 2);
    }
    if (cur - 1 > 0 && dist[cur - 1] == -1) {
      dist[cur - 1] = dist[cur] + 1;
      q.push(cur - 1);
    }
  }
  return 0;
}

树上 DFS 带状态

适用：树上统计叶子、路径限制、从根到点维护一个当前状态。

易错点：树可以不用 visited，用 parent 防止走回父节点；状态要作为参数传下去；叶子判断不能把父节点算作儿子。


#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
vector<int> adj[N];
int cat[N];
int n,m,ans;
void dfs(int u,int fa,int cnt) {
  if (cat[u]) cnt++;
  else cnt = 0;
  if (cnt > m) return;
  bool isLeaf = true;
  for (int v : adj[u]) {
    if (v == fa) continue;
    isLeaf = false;
    dfs(v,u,cnt);
  }
  if (isLeaf) ans++;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> m;
  for (int i = 1;i <= n;i++) cin >> cat[i];
  for (int i = 1;i < n;i++) {
    int u,v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1,0,0);
  cout << ans << endl;
  return 0;
}

并查集 DSU

适用：动态合并集合、判断连通块、Kruskal、关系分组。

易错点：每次多测要 init；find 里路径压缩；需要连通块大小时维护 sz。


#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int pre[N],sz[N];
void init(int n) {
  for (int i = 1;i <= n;i++) {
    pre[i] = i;
    sz[i] = 1;
  }
}
int find(int x) {
  if (pre[x] == x) return x;
  return pre[x] = find(pre[x]);
}
void join(int x,int y) {
  int fx = find(x);
  int fy = find(y);
  if (fx == fy) return;
  pre[fy] = fx;
  sz[fx] += sz[fy];
}

Kruskal 最小生成树

适用：无向图，要求连接所有点的最小代价；边数不太离谱时优先用。

易错点：边排序；只选连接两个不同连通块的边；最后边数应为 n - 1。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5 + 10;
int pre[N];
struct Edge {
  int u,v,w;
};
void init(int n) {
  for (int i = 1;i <= n;i++) pre[i] = i;
}
int find(int x) {
  if (pre[x] == x) return x;
  return pre[x] = find(pre[x]);
}
bool join(int x,int y) {
  int fx = find(x);
  int fy = find(y);
  if (fx == fy) return false;
  pre[fy] = fx;
  return true;
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  init(n);
  vector<Edge> edges;
  for (int i = 1;i <= m;i++) {
    int u,v,w;
    cin >> u >> v >> w;
    edges.push_back({u,v,w});
  }
  sort(edges.begin(),edges.end(),[](const Edge& a,const Edge& b) {
    return a.w < b.w;
  });
  int ans = 0,cnt = 0;
  for (Edge e : edges) {
    if (join(e.u,e.v)) {
      ans += e.w;
      cnt++;
    }
  }
  if (cnt != n - 1) cout << "orz" << endl;
  else cout << ans << endl;
  return 0;
}

Prim O(n^2)

适用：完全图/隐式完全图 MST，比如 1245D 这种不用真的建所有边的题。

易错点：minDist 初始含义要明确；每轮选未访问的最小点；选中后用它更新其他点。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2010;
const int INF = 2e18;
int c[N],k[N];
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<pair<int,int>> city(n + 1);
  for (int i = 1;i <= n;i++) cin >> city[i].first >> city[i].second;
  for (int i = 1;i <= n;i++) cin >> c[i];
  for (int i = 1;i <= n;i++) cin >> k[i];
  vector<int> minDist(n + 1),parent(n + 1,0);
  vector<bool> vis(n + 1,false);
  for (int i = 1;i <= n;i++) minDist[i] = c[i];
  int ans = 0;
  vector<int> stations;
  vector<pair<int,int>> connections;
  for (int t = 1;t <= n;t++) {
    int cur = -1,minVal = INF;
    for (int i = 1;i <= n;i++) {
      if (!vis[i] && minDist[i] < minVal) {
        minVal = minDist[i];
        cur = i;
      }
    }
    vis[cur] = true;
    ans += minVal;
    if (parent[cur] == 0) stations.push_back(cur);
    else connections.push_back({cur,parent[cur]});
    for (int i = 1;i <= n;i++) {
      if (vis[i]) continue;
      int cost = (k[cur] + k[i]) *
      (abs(city[cur].first - city[i].first) + abs(city[cur].second - city[i].second));
      if (cost < minDist[i]) {
        minDist[i] = cost;
        parent[i] = cur;
      }
    }
  }
  cout << ans << endl;
  return 0;
}

Dijkstra 最短路

适用：单源最短路，边权非负。需要路径时维护 parent。

易错点：有负边不能用；priority_queue 里是 pair<dist,node>；vis 出堆后标记。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int INF = 2e18;
struct Edge {
  int to;
  int weight;
};
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  vector<vector<Edge>> graph(n + 1);
  for (int i = 1;i <= m;i++) {
    int u,v,w;
    cin >> u >> v >> w;
    graph[u].push_back({v,w});
    graph[v].push_back({u,w});
  }
  vector<int> minDist(n + 1,INF);
  vector<int> parent(n + 1,-1);
  vector<bool> vis(n + 1,false);
  priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>> pq;
  minDist[1] = 0;
  pq.push({0,1});
  while (!pq.empty()) {
    int cur = pq.top().second;
    pq.pop();
    if (vis[cur]) continue;
    vis[cur] = true;
    for (auto edge : graph[cur]) {
      int to = edge.to;
      int weight = edge.weight;
      if (!vis[to] && minDist[cur] + weight < minDist[to]) {
        minDist[to] = minDist[cur] + weight;
        parent[to] = cur;
        pq.push({minDist[to],to});
      }
    }
  }
  if (minDist[n] == INF) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> path;
  for (int cur = n;cur != -1;cur = parent[cur]) path.push_back(cur);
  reverse(path.begin(),path.end());
  for (int x : path) cout << x << " ";
  cout << endl;
  return 0;
}

Floyd 多源最短路

适用：点数较小，要求任意两点最短路；可以处理负边，但不能有负环。

易错点：初始化 dist[i][i] = 0；重边取最小；三重循环顺序必须是 k,i,j；INF 不要相加溢出。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int INF = 2e18;
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  vector<vector<int>> dist(n + 1,vector<int>(n + 1,INF));
  for (int i = 1;i <= n;i++) {
    dist[i][i] = 0;
  }
  for (int i = 1;i <= m;i++) {
    int u,v,w;
    cin >> u >> v >> w;
    dist[u][v] = min(dist[u][v],w);
    // 无向图再加这一句
    // dist[v][u] = min(dist[v][u],w);
  }
  for (int k = 1;k <= n;k++) {
    for (int i = 1;i <= n;i++) {
      for (int j = 1;j <= n;j++) {
        if (dist[i][k] == INF || dist[k][j] == INF) continue;
        dist[i][j] = min(dist[i][j],dist[i][k] + dist[k][j]);
      }
    }
  }
  int q;
  cin >> q;
  while (q--) {
    int u,v;
    cin >> u >> v;
    if (dist[u][v] == INF) cout << -1 << endl;
    else cout << dist[u][v] << endl;
  }
  return 0;
}

拓扑排序

适用：有向无环图、依赖关系、先后顺序、判环。

易错点：重复边可能导致入度重复；出队数量小于 n 表示有环；字符串建图要处理前缀非法。


#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
vector<int> g[N];
int in[N];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  for (int i = 1;i <= m;i++) {
    int u,v;
    cin >> u >> v;
    g[u].push_back(v);
    in[v]++;
  }
  queue<int> q;
  for (int i = 1;i <= n;i++) {
    if (in[i] == 0) q.push(i);
  }
  vector<int> order;
  while (!q.empty()) {
    int cur = q.front();
    q.pop();
    order.push_back(cur);
    for (int v : g[cur]) {
      in[v]--;
      if (in[v] == 0) q.push(v);
    }
  }
  if ((int)order.size() < n) {
    cout << "Cycle" << endl;
  }
  else {
    for (int x : order) cout << x << " ";
    cout << endl;
  }
  return 0;
}

Floyd 全源最短路

适用：n 较小，求任意两点最短路；也常用于加边/加点后的最短路变形。

复杂度：O(n^3)，通常 n <= 500 左右才考虑。

易错点：初始化 d[i][i] = 0；重边取 min；INF 不能太小；循环顺序必须 k 在最外层。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int INF = 2e18;
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  vector<vector<int>> d(n + 1,vector<int>(n + 1,INF));
  for (int i = 1;i <= n;i++) d[i][i] = 0;
  for (int i = 1;i <= m;i++) {
    int u,v,w;
    cin >> u >> v >> w;
    d[u][v] = min(d[u][v],w);
    d[v][u] = min(d[v][u],w);
  }
  for (int k = 1;k <= n;k++) {
    for (int i = 1;i <= n;i++) {
      for (int j = 1;j <= n;j++) {
        if (d[i][k] == INF || d[k][j] == INF) continue;
        d[i][j] = min(d[i][j],d[i][k] + d[k][j]);
      }
    }
  }
  return 0;
}

Core Techniques

二分答案/位置

适用：有序数组查位置，或答案具有单调性。

易错点：lower_bound 找第一个 >= x；upper_bound 找第一个 > x；自己写二分时边界要统一。


// 前缀和上找第一个 >= x 的位置
vector<int> pre(n + 1);
for (int i = 1;i <= n;i++) pre[i] = pre[i - 1] + a[i];
int pos = lower_bound(pre.begin() + 1,pre.end(),x) - pre.begin();

二分答案模板

适用：答案具有单调性，比如最大化最小值、最小化最大值、能否在 x 限制内完成。

易错点：check(mid) 的含义必须固定；最大化答案用 l = mid，最小化答案用 r = mid；注意 mid 写法防死循环。


bool check(int x) {
  // 如果 x 可行，返回 true
  return true;
}
int l = 0,r = 1e18;
while (l < r) {
  int mid = (l + r + 1) / 2;
  if (check(mid)) l = mid;
  else r = mid - 1;
}
cout << l << endl;

前缀和/差分

适用：多次区间查询用前缀和；多次区间加用差分。

易错点：差分更新 [l,r] 是 diff[l]++、diff[r+1]--；查询 [l,r] 是 pre[r]-pre[l-1]。


// 前缀和
vector<int> pre(n + 1);
for (int i = 1;i <= n;i++) pre[i] = pre[i - 1] + a[i];
int sum = pre[r] - pre[l - 1];
// 差分
diff[l]++;
diff[r + 1]--;
for (int i = 1;i <= N;i++) diff[i] += diff[i - 1];

双指针/滑动窗口

适用：连续区间，右端点右移，左端点只会右移，条件具有单调性。

易错点：while 收缩条件要写清楚；答案更新位置要在窗口合法后。


int ans = 0;
int l = 1;
int sum = 0;
for (int r = 1;r <= n;r++) {
  sum += a[r];
  while (sum > t) {
    sum -= a[l];
    l++;
  }
  ans = max(ans,r - l + 1);
}
cout << ans << endl;

单调栈

适用：找每个位置左/右第一个比它大或小的元素。

易错点：栈里通常存下标；维护递增栈还是递减栈要先确定；相等时弹不弹取决于题意。


// 找每个 i 左边第一个严格小于 a[i] 的位置
vector<int> leftLess(n + 1,0);
stack<int> st;
for (int i = 1;i <= n;i++) {
  while (!st.empty() && a[st.top()] >= a[i]) {
    st.pop();
  }
  if (!st.empty()) leftLess[i] = st.top();
  else leftLess[i] = 0;
  st.push(i);
}

单调队列

适用：固定长度滑动窗口最大值/最小值，或 DP 中维护区间最值。

易错点：队列存下标；先弹出过期下标，再维护单调性；窗口形成后再记录答案。


// 每个长度为 k 的窗口最大值
deque<int> q;
for (int i = 1;i <= n;i++) {
  while (!q.empty() && q.front() <= i - k) {
    q.pop_front();
  }
  while (!q.empty() && a[q.back()] <= a[i]) {
    q.pop_back();
  }
  q.push_back(i);
  if (i >= k) {
    cout << a[q.front()] << " ";
  }
}
cout << endl;

树状数组 Fenwick

适用：单点修改 + 前缀和查询；也可做逆序对、动态排名的基础工具。

复杂度：修改和查询都是 O(log n)。

易错点：下标从 1 开始；lowbit 写 x & -x；区间和是 query(r) - query(l - 1)。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5 + 10;
int tr[N];
int lowbit(int x) {
  return x & -x;
}
void add(int x,int v) {
  for (int i = x;i < N;i += lowbit(i)) {
    tr[i] += v;
  }
}
int query(int x) {
  int res = 0;
  for (int i = x;i > 0;i -= lowbit(i)) {
    res += tr[i];
  }
  return res;
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,q;
  cin >> n >> q;
  for (int i = 1;i <= n;i++) {
    int x;
    cin >> x;
    add(i,x);
  }
  while (q--) {
    int op,l,r;
    cin >> op >> l >> r;
    if (op == 1) {
      add(l,r);
    }
    else {
      cout << query(r) - query(l - 1) << endl;
    }
  }
  return 0;
}

Data Structures and DP

离散化

适用：值域很大，但只关心大小关系/排名/出现次数，需要把 1e9 或 1e18 的数压成 1..cnt。常和树状数组、计数、DP 搭配。

易错点：离散化只保持大小关系；如果题目关心真实距离或区间长度，不能直接用离散后的编号差代替原值差。


#include <bits/stdc++.h>
using namespace std;
#define int long long
vector<int> alls;
int getId(int x) {
  return lower_bound(alls.begin(),alls.end(),x) - alls.begin() + 1;
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n + 1);
  for (int i = 1;i <= n;i++) {
    cin >> a[i];
    alls.push_back(a[i]);
  }
  sort(alls.begin(),alls.end());
  alls.erase(unique(alls.begin(),alls.end()),alls.end());
  for (int i = 1;i <= n;i++) {
    int id = getId(a[i]);
    cout << id << " ";
  }
  cout << endl;
  return 0;
}

0/1 背包

适用：每个物品最多选一次，容量限制下求最大值/可行性/方案数。

易错点：一维优化时容量必须倒序，防止同一个物品被重复使用。


vector<int> dp(V + 1,0);
for (int i = 1;i <= n;i++) {
  for (int j = V;j >= w[i];j--) {
    dp[j] = max(dp[j],dp[j - w[i]] + val[i]);
  }
}
cout << dp[V] << endl;

完全背包

适用：每个物品可以选无限次。

易错点：一维优化时容量正序，因为当前物品可以重复用。


vector<int> dp(V + 1,0);
for (int i = 1;i <= n;i++) {
  for (int j = w[i];j <= V;j++) {
    dp[j] = max(dp[j],dp[j - w[i]] + val[i]);
  }
}
cout << dp[V] << endl;

线性 DP / 打家劫舍模型

适用：选了当前状态会影响相邻状态，比如 455A。

易错点：先把题目转成按值域/位置的阶段；初始化 dp[0]、dp[1]。


// dp[i] 表示只考虑 1..i 时能得到的最大值
dp[0] = 0;
dp[1] = cnt[1];
for (int i = 2;i <= Max;i++) {
  dp[i] = max(dp[i - 1],dp[i - 2] + cnt[i]);
}
cout << dp[Max] << endl;

状态机 DP

适用：每一天/每一列有几种互斥状态，比如休息、选上、选下、不选。

易错点：先定义 dp[i][state] 的含义，再枚举上一状态转移。


// 例：每列选上、选下、不选，不能和上一列同一行相邻
vector<vector<int>> dp(n + 1,vector<int>(3,0));
for (int i = 1;i <= n;i++) {
  dp[i][0] = max({dp[i - 1][0],dp[i - 1][1],dp[i - 1][2]});
  dp[i][1] = max(dp[i - 1][0],dp[i - 1][2]) + up[i];
  dp[i][2] = max(dp[i - 1][0],dp[i - 1][1]) + down[i];
}
cout << max({dp[n][0],dp[n][1],dp[n][2]}) << endl;

余数 DP / 模数可行性 DP

适用：问能不能选一些数，使它们的和对 m 取模后等于某个余数，典型题是 577B。

复杂度：普通写法是 O(n * m) 时间、O(m) 空间。n 和 m 同时接近 1e5 时不能直接跑 DP；只有当题目能用鸽巢原理推出 n >= m 直接 YES 时，才可以承受这种数据。

状态：dp[r] 表示当前处理过的数里，能否选出一个非空子序列，使得 sum % m == r。

易错点：非空子序列不能初始化 dp[0] = true；每个数只能用一次，所以每轮用 ndp 从旧 dp 转移；如果 n >= m，可以用鸽巢原理直接 YES。


#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  vector<int> a(n + 1);
  for (int i = 1;i <= n;i++) {
    cin >> a[i];
    a[i] %= m;
  }
  if (n >= m) {
    cout << "YES" << endl;
    return 0;
  }
  vector<int> dp(m,0);
  for (int i = 1;i <= n;i++) {
    vector<int> ndp = dp;
    ndp[a[i]] = 1;
    for (int r = 0;r < m;r++) {
      if (dp[r]) {
        ndp[(r + a[i]) % m] = 1;
      }
    }
    dp = ndp;
  }
  if (dp[0]) cout << "YES" << endl;
  else cout << "NO" << endl;
  return 0;
}

余数 DP bitset 优化版

适用：m 最大约 1e5，普通 O(n * m) 余数 DP 太慢时使用。复杂度约 O(n * m / 64)，空间 O(m)。

限制：这不是 O(n) 算法；如果 n 和 m 都是 1e5，仍然可能比较吃时间，但比普通双循环可用得多。MAXM 要大于题目里的最大 m。

核心：dp[r] 表示余数 r 是否可达，加入 v 后相当于把所有可达余数循环右移 v 位。


#include <bits/stdc++.h>
using namespace std;
const int MAXM = 100000 + 10;
bitset<MAXM> dp,maskBits;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,m;
  cin >> n >> m;
  for (int i = 0;i < m;i++) {
    maskBits[i] = 1;
  }
  for (int i = 1;i <= n;i++) {
    int x;
    cin >> x;
    int v = x % m;
    if (v == 0) {
      cout << "YES" << endl;
      return 0;
    }
    bitset<MAXM> shifted = (dp << v) | (dp >> (m - v));
    shifted &= maskBits;
    dp |= shifted;
    dp[v] = 1;
    if (dp[0]) {
      cout << "YES" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
  return 0;
}

Strings, Trees and Math

KMP 字符串匹配

适用：在文本串 s 中找模式串 p 的出现次数/位置，复杂度 O(n + m)。

易错点：next 数组表示 p[0..i] 的最长相等真前后缀长度；匹配失败时 j = ne[j - 1]；匹配成功后统计多次出现要继续 j = ne[j - 1]。


#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  string s,p;
  cin >> s >> p;
  int n = s.size();
  int m = p.size();
  vector<int> ne(m,0);
  for (int i = 1,j = 0;i < m;i++) {
    while (j > 0 && p[i] != p[j]) {
      j = ne[j - 1];
    }
    if (p[i] == p[j]) {
      j++;
    }
    ne[i] = j;
  }
  vector<int> pos;
  for (int i = 0,j = 0;i < n;i++) {
    while (j > 0 && s[i] != p[j]) {
      j = ne[j - 1];
    }
    if (s[i] == p[j]) {
      j++;
    }
    if (j == m) {
      pos.push_back(i - m + 1);
      j = ne[j - 1];
    }
  }
  cout << pos.size() << endl;
  for (int x : pos) {
    cout << x << " ";
  }
  cout << endl;
  return 0;
}

LCA 倍增

适用：树上最近公共祖先、树上两点距离、路径查询基础。

复杂度：预处理 O(n log n)，每次查询 O(log n)。

易错点：LOG 要够大；根节点父亲设为 0；先把深度较大的点跳到同一深度；最后两个点一起向上跳。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5 + 10;
const int LOG = 20;
vector<pair<int,int>> g[N];
int depth[N];
int fa[N][LOG + 1];
int distRoot[N];
void dfs(int u,int father) {
  fa[u][0] = father;
  for (int i = 1;i <= LOG;i++) {
    fa[u][i] = fa[fa[u][i - 1]][i - 1];
  }
  for (auto edge : g[u]) {
    int v = edge.first;
    int w = edge.second;
    if (v == father) continue;
    depth[v] = depth[u] + 1;
    distRoot[v] = distRoot[u] + w;
    dfs(v,u);
  }
}
int lca(int a,int b) {
  if (depth[a] < depth[b]) swap(a,b);
  for (int i = LOG;i >= 0;i--) {
    if (depth[fa[a][i]] >= depth[b]) {
      a = fa[a][i];
    }
  }
  if (a == b) return a;
  for (int i = LOG;i >= 0;i--) {
    if (fa[a][i] != fa[b][i]) {
      a = fa[a][i];
      b = fa[b][i];
    }
  }
  return fa[a][0];
}
int getDist(int a,int b) {
  int c = lca(a,b);
  return distRoot[a] + distRoot[b] - 2 * distRoot[c];
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n,q;
  cin >> n >> q;
  for (int i = 1;i < n;i++) {
    int u,v,w;
    cin >> u >> v >> w;
    g[u].push_back({v,w});
    g[v].push_back({u,w});
  }
  depth[1] = 1;
  dfs(1,0);
  while (q--) {
    int a,b;
    cin >> a >> b;
    cout << lca(a,b) << " " << getDist(a,b) << endl;
  }
  return 0;
}

基础数学

适用：gcd、快速幂、取模、质数筛。省赛签到和中低档数论常用。


int qpow(int a,int b,int mod) {
  int res = 1 % mod;
  while (b) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
    b >>= 1;
  }
  return res;
}
int gcd(int a,int b) {
  return b == 0 ? a : gcd(b,a % b);
}
vector<int> primes;
bool st[N];
void get_primes(int n) {
  for (int i = 2;i <= n;i++) {
    if (!st[i]) primes.push_back(i);
    for (int p : primes) {
      if (i * p > n) break;
      st[i * p] = true;
      if (i % p == 0) break;
    }
  }
}

组合数和排列数

适用：多次查询 C(n,k)、A(n,k)，并且取模 MOD 为质数。常用于方案数题。

复杂度：预处理 O(N)，每次查询 O(1)。

易错点：k < 0 或 k > n 时答案是 0；MOD 必须是质数才能用费马小定理求逆元。


#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 1e6 + 10;
const int MOD = 1e9 + 7;
int fac[N],invFac[N];
int qpow(int a,int b) {
  int res = 1;
  while (b) {
    if (b & 1) res = res * a % MOD;
    a = a * a % MOD;
    b >>= 1;
  }
  return res;
}
void initComb(int n) {
  fac[0] = 1;
  for (int i = 1;i <= n;i++) {
    fac[i] = fac[i - 1] * i % MOD;
  }
  invFac[n] = qpow(fac[n],MOD - 2);
  for (int i = n - 1;i >= 0;i--) {
    invFac[i] = invFac[i + 1] * (i + 1) % MOD;
  }
}
int C(int n,int k) {
  if (k < 0 || k > n) return 0;
  return fac[n] * invFac[k] % MOD * invFac[n - k] % MOD;
}
int A(int n,int k) {
  if (k < 0 || k > n) return 0;
  return fac[n] * invFac[n - k] % MOD;
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  initComb(1000000);
  int n,k;
  cin >> n >> k;
  cout << C(n,k) << " " << A(n,k) << endl;
  return 0;
}

Pre-contest Checklist

  1. 1. long long 是否需要。
  2. 2. 多测时数组、vector、入度、vis 是否清空。
  3. 3. 图是否连通；最短路是否有负边。
  4. 4. BFS/DFS 是否越界，visited 是否及时标记。
  5. 5. 背包 0/1 倒序，完全背包正序。
  6. 6. 拓扑排序出队数量是否等于 n。
  7. 7. 二分边界和 lower_bound/upper_bound 含义是否用对。
